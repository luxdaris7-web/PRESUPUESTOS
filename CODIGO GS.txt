// ---------------------------------------------------------------
//  ENTRADA DE LA API
// ---------------------------------------------------------------
function doGet() {
  return json_({ ok: true, mensaje: 'API de presupuestos activa' });
}

function doPost(e) {
  try {
    const p = JSON.parse((e && e.postData && e.postData.contents) || '{}');

    // Recuperar clave: va ANTES de validar, porque quien la olvidó no puede enviarla
    if (p.accion === 'recuperarClave') return json_({ ok: true, data: recuperarClave_() });

    if (p.clave !== CONFIG.CLAVE_ACCESO) return json_({ ok: false, error: 'CLAVE_INVALIDA' });

    const acciones = {
      arranque:          () => obtenerDatosDeArranque(),
      generarPdf:        () => generarPdf(p.datos),
      enviarCorreo:      () => enviarPresupuestoPorCorreo(p.datos, p.correo),
      guardar:           () => guardarEnHistorial(p.datos),
      historial:         () => obtenerHistorial(),
      detalle:           () => obtenerDetalle(p.numero),
      reimprimir:        () => reimprimirPresupuesto(p.numero),
      clientes:          () => obtenerClientes(),
      guardarCliente:    () => guardarClienteBD(p.cliente),
      actualizarCliente: () => actualizarCliente(p.fila, p.valores)
    };

    const fn = acciones[p.accion];
    if (!fn) throw new Error('Acción desconocida: ' + p.accion);
    return json_({ ok: true, data: fn() });
  } catch (err) {
    return json_({ ok: false, error: err.message });
  }
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }

function hojaPresupuesto_() {
  const ss = ss_();
  const hoja = ss.getSheetByName(CONFIG.HOJA_PRESUPUESTO);
  if (hoja) return hoja;
  // Respaldo: la primera pestaña que no sea Historial ni Base_Clientes
  const otra = ss.getSheets().find(h => [CONFIG.HOJA_HISTORIAL, CONFIG.HOJA_CLIENTES].indexOf(h.getName()) === -1);
  if (!otra) throw new Error('No encuentro la hoja del presupuesto. Revisa CONFIG.HOJA_PRESUPUESTO.');
  return otra;
}

function conBloqueo_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000); // evita que dos presupuestos se escriban a la vez en la hoja
  try { return fn(); } finally { lock.releaseLock(); }
}

function numeroEntero_(valor) {
  return parseInt(String(valor).replace(/\D/g, ''), 10) || 0;
}

function formatearNumero_(valor) {
  const n = numeroEntero_(valor) || 1;
  return ('00000' + n).slice(-5);
}

function redondear_(n) { return Math.round(n * 100) / 100; }

function normalizarDatos_(d) {
  if (!d) throw new Error('No llegaron datos del presupuesto.');
  const items = (d.items || [])
    .map(it => ({ desc: String(it.desc || '').trim(), cant: Number(it.cant) || 0, precio: Number(it.precio) || 0 }))
    .filter(it => it.desc);
  if (!items.length) throw new Error('El presupuesto no tiene artículos.');
  return {
    numero: formatearNumero_(d.numero),
    cliente: String(d.cliente || '').trim(),
    rif: String(d.rif || '').trim(),
    correo: String(d.correo || '').trim(),
    fecha: String(d.fecha || '').trim(),
    vigencia: Number(d.vigencia) || 0,
    tasa: Number(d.tasa) || 0,
    iva: Number(d.iva) || 0,
    items: items
  };
}

function calcularTotales_(d) {
  const sub = d.items.reduce((s, it) => s + it.cant * it.precio, 0);
  const iva = sub * d.iva / 100;
  const total = sub + iva;
  return { sub: redondear_(sub), iva: redondear_(iva), total: redondear_(total), totalBs: redondear_(total * d.tasa) };
}

// ---------------------------------------------------------------
//  1. DATOS DE ARRANQUE
// ---------------------------------------------------------------
function obtenerDatosDeArranque() {
  const hc = ss_().getSheetByName(CONFIG.HOJA_CLIENTES);
  const clientes = [];
  if (hc) {
    const data = hc.getDataRange().getDisplayValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][1]) clientes.push({ nombre: data[i][1], rif: data[i][2], telefono: data[i][3], correo: data[i][4] });
    }
  }
  return { numero: proximoNumero_(), clientes: clientes };
}

function proximoNumero_() {
  const h = ss_().getSheetByName(CONFIG.HOJA_HISTORIAL);
  let max = 0;
  if (h && h.getLastRow() > 1) {
    h.getRange(2, 1, h.getLastRow() - 1, 1).getValues().forEach(r => {
      const n = numeroEntero_(r[0]);
      if (n > max) max = n;
    });
  }
  return formatearNumero_(max + 1);
}

// ---------------------------------------------------------------
//  2. ESCRIBIR EL PRESUPUESTO EN LA HOJA
//     Fila 7: título · Fila 9: encabezados · Filas 10-45: artículos
//     Filas 48-53: cliente + resumen · Filas 54-55: dirección y teléfonos
//     Filas 49-53: sello digital (flota encima del recuadro, centrado).
//     Devuelve la última fila usada para saber dónde cortar el PDF.
// ---------------------------------------------------------------
function escribirDatosEnHoja_(d) {
  const hoja = hojaPresupuesto_();
  const F = CONFIG.FILA_ITEMS;
  const n = d.items.length;
  const t = calcularTotales_(d);
  const SOLIDO = SpreadsheetApp.BorderStyle.SOLID;
  const MEDIO = SpreadsheetApp.BorderStyle.SOLID_MEDIUM;

  // Altura (px) de las filas 1-9 (membrete + título + encabezado): punto de partida del reparto en páginas
  let yInicio = 0;
  for (let r = 1; r < F; r++) {
    yInicio += r === CONFIG.FILA_TITULO ? 30 : (r === CONFIG.FILA_ENCABEZADO ? 22 : hoja.getRowHeight(r));
  }

  // Reparto de artículos: hasta 36 caben en una página con el recuadro anclado en la fila 48.
  // Con más, la tabla sigue en la página siguiente (repitiendo el encabezado) y el recuadro con los
  // totales, la dirección y el sello queda completo en la ÚLTIMA página, nunca partido.
  const plan = planificarFilas_(d, yInicio, anchoTabla_(hoja));
  const filasTabla = plan.filas;
  Logger.log('Páginas planificadas: ' + plan.paginas);
  const s = F + filasTabla.length;
  const filaDir = s + 6;                    // 54
  const filaTel = s + 7;                    // 55
  const filaSello = s + 1;                  // 49: primera fila de datos del recuadro (el sello va encima)
  const ultimaFila = filaTel;               // 55

  // Quitar protecciones que impidan escribir desde la fila 6 (A:D). El membrete (filas 1-5) queda como esté.
  liberarZonaProtegida_(hoja);

  // Asegurar filas suficientes
  if (hoja.getMaxRows() < ultimaFila + 1) hoja.insertRowsAfter(hoja.getMaxRows(), ultimaFila + 1 - hoja.getMaxRows());

  // Quitar el sello anterior (solo el sello; el logo del membrete no se toca)
  hoja.getImages().forEach(img => { if (img.getAltTextTitle() === 'SELLO_DIGITAL') img.remove(); });

  // Limpiar todo lo que hay desde la fila 6 hacia abajo (columnas A:D)
  const L = CONFIG.FILA_LIMPIEZA;
  const zona = hoja.getRange(L, 1, hoja.getMaxRows() - L + 1, 4);
  zona.breakApart();
  zona.clear();
  // Alturas limpias (21 px) desde los artículos hasta el final: evita que queden alturas de un presupuesto anterior
  hoja.setRowHeights(F, ultimaFila - F + 1, 21);

  // --- Fila 7: PRESUPUESTO + número ---
  hoja.getRange(CONFIG.FILA_TITULO, 1, 1, 4).merge()
    .setValue('PRESUPUESTO ' + d.numero)
    .setFontFamily('Arial').setFontSize(14).setFontWeight('bold')
    .setHorizontalAlignment('center').setVerticalAlignment('middle');
  hoja.setRowHeight(CONFIG.FILA_TITULO, 30);

  // --- Fila 9: encabezados de la tabla ---
  hoja.getRange(CONFIG.FILA_ENCABEZADO, 1, 1, 4)
    .setValues([['DESCRIPCION', 'CANTIDAD', 'PRECIO UNITARIO', 'SUBTOTAL']])
    .setFontFamily('Arial').setFontSize(9).setFontWeight('bold')
    .setHorizontalAlignment('center').setVerticalAlignment('middle')
    .setBorder(true, true, true, true, true, true, 'black', SOLIDO);
  hoja.setRowHeight(CONFIG.FILA_ENCABEZADO, 22);

  // --- Artículos (desde la fila 10). Si la tabla pasa a otra página, esa página empieza con el encabezado repetido ---
  const ENCABEZADO = ['DESCRIPCION', 'CANTIDAD', 'PRECIO UNITARIO', 'SUBTOTAL'];
  const valores = filasTabla.map((f, k) => {
    const r = F + k;
    if (f.tipo === 'item') {
      const it = d.items[f.i];
      return [it.desc, it.cant, it.precio, '=B' + r + '*C' + r];
    }
    return f.tipo === 'enc' ? ENCABEZADO : ['', '', '', ''];
  });
  hoja.getRange(F, 1, valores.length, 4).setValues(valores)
    .setFontFamily('Arial').setFontSize(9).setVerticalAlignment('middle');

  // Formato por tramos seguidos del mismo tipo (artículos / encabezado repetido / filas en blanco)
  let k = 0;
  while (k < filasTabla.length) {
    const tipo = filasTabla[k].tipo;
    let j = k;
    while (j + 1 < filasTabla.length && filasTabla[j + 1].tipo === tipo) j++;
    const cnt = j - k + 1, fila0 = F + k;
    if (tipo === 'item') {
      hoja.getRange(fila0, 1, cnt, 4).setBorder(true, true, true, true, true, true, 'black', SOLIDO);
      hoja.getRange(fila0, 1, cnt, 1).setWrap(true).setHorizontalAlignment('left');
      hoja.getRange(fila0, 2, cnt, 1).setHorizontalAlignment('center').setNumberFormat('General');
      hoja.getRange(fila0, 3, cnt, 2).setHorizontalAlignment('center').setNumberFormat(CONFIG.FMT_USD);
    } else if (tipo === 'enc') {
      hoja.getRange(fila0, 1, cnt, 4).setFontWeight('bold')
        .setHorizontalAlignment('center')
        .setBorder(true, true, true, true, true, true, 'black', SOLIDO);
    }
    k = j + 1;
  }
  // Alto de cada fila según el largo de la descripción (las demás ya quedaron en 21)
  filasTabla.forEach((f, i) => { if (f.h !== 21) hoja.setRowHeight(F + i, f.h); });

  // --- Recuadro DATOS DEL CLIENTE / RESUMEN (filas 48 a 53) ---
  const izquierda = [
    'CLIENTE:  ' + (d.cliente || '—'),
    'RIF / CI:  ' + (d.rif || '—'),
    'Nº PRESUPUESTO:  ' + d.numero,
    'FECHA:  ' + (d.fecha || '—'),
    'VIGENCIA:  ' + d.vigencia + ' Días'
  ];
  const derecha = [
    ['SUB TOTAL', t.sub, CONFIG.FMT_USD],
    ['IVA (' + d.iva + '%)', t.iva, CONFIG.FMT_USD],
    ['TOTAL PRESUPUESTO', t.total, CONFIG.FMT_USD],
    ['TASA BCV', d.tasa, CONFIG.FMT_BS],
    ['TOTAL EN BS', t.totalBs, CONFIG.FMT_BS]
  ];

  const bloque = hoja.getRange(s, 1, 6, 4);
  bloque.setFontFamily('Arial').setFontSize(9).setFontWeight('bold').setVerticalAlignment('middle')
    .setBackground(CONFIG.AMARILLO);

  hoja.getRange(s, 1).setValue('DATOS DEL CLIENTE');
  hoja.getRange(s, 2, 1, 3).merge().setValue('RESUMEN');
  hoja.getRange(s, 1, 1, 4).setBackground(CONFIG.VERDE).setHorizontalAlignment('center');

  izquierda.forEach((txt, i) => {
    hoja.getRange(s + 1 + i, 1).setValue(txt).setWrap(true).setHorizontalAlignment('left');
  });
  derecha.forEach((f, i) => {
    const r = s + 1 + i;
    hoja.getRange(r, 2, 1, 2).merge().setValue(f[0]).setHorizontalAlignment('right');
    hoja.getRange(r, 4).setValue(f[1]).setNumberFormat(f[2]).setHorizontalAlignment('center');
  });

  // Resaltar los dos totales
  hoja.getRange(s + 3, 2, 1, 3).setBackground(CONFIG.VERDE);
  hoja.getRange(s + 5, 2, 1, 3).setBackground(CONFIG.VERDE);

  bloque.setBorder(true, true, true, true, true, true, 'black', SOLIDO);
  hoja.setRowHeights(s, 6, 21);

  // --- Filas 54 y 55: dirección y teléfonos (letra azul, fondo blanco, recuadro) ---
  hoja.getRange(filaDir, 1, 1, 4).merge().setValue(CONFIG.DIRECCION);
  hoja.getRange(filaTel, 1, 1, 4).merge().setValue(CONFIG.TELEFONOS);
  hoja.getRange(filaDir, 1, 2, 4)
    .setFontFamily('Arial').setFontSize(11).setFontWeight('bold').setFontColor(CONFIG.COLOR_AZUL)
    .setBackground('#ffffff').setHorizontalAlignment('center').setVerticalAlignment('middle')
    .setBorder(true, true, true, true, false, false, 'black', MEDIO);
  hoja.setRowHeights(filaDir, 2, 24);

  // Guardar TODO lo escrito antes de tocar el sello, para que un fallo de imagen no deshaga nada
  SpreadsheetApp.flush();

  // --- Sello digital encima del recuadro, centrado (si falla, el PDF sale igual, sin sello) ---
  try {
    insertarSello_(hoja, filaSello, CONFIG.FILAS_SELLO);
  } catch (err) {
    Logger.log('SELLO NO INSERTADO: ' + err.message);
  }

  SpreadsheetApp.flush();
  return ultimaFila;
}

// Ancho total (px) de las columnas A:D
function anchoTabla_(hoja) {
  let total = 0;
  for (let c = 1; c <= 4; c++) total += hoja.getColumnWidth(c);
  return total;
}

// Alto de la fila de un artículo según el largo de la descripción (aprox. 58 caracteres por línea)
function alturaItem_(it) {
  const lineas = Math.max(1, Math.ceil(it.desc.length / 58));
  return Math.max(21, lineas * 13 + 8);
}

// Decide qué va en cada fila desde la 10 hasta justo antes del recuadro de totales.
// Devuelve { filas: [{tipo:'item'|'enc'|'blanco', h, i}], paginas }.
//  · Hasta 36 artículos: artículos + filas en blanco hasta la fila 47 (recuadro en la 48, una sola página).
//  · Más de 36: se calcula dónde termina cada página; al llegar al final se rellena con filas en blanco
//    hasta pasar el corte, y la página nueva arranca con el encabezado de la tabla repetido.
//    Antes del recuadro se hace lo mismo si este no cabe completo (totales + dirección + teléfonos),
//    así el recuadro nunca queda partido entre dos páginas.
function planificarFilas_(d, yInicio, ancho) {
  const F = CONFIG.FILA_ITEMS;
  const filas = [];

  if (F + d.items.length + CONFIG.FILAS_LIBRES <= CONFIG.FILA_RESUMEN) {
    d.items.forEach((it, i) => filas.push({ tipo: 'item', h: alturaItem_(it), i: i }));
    while (F + filas.length < CONFIG.FILA_RESUMEN) filas.push({ tipo: 'blanco', h: 21 });
    return { filas: filas, paginas: 1 };
  }

  const U = ancho * (CONFIG.PAGINA_ALTO_IN / CONFIG.PAGINA_ANCHO_IN); // alto de una página, en px de la hoja
  const margen = U * CONFIG.PAGINA_MARGEN;
  const H_ENC = 22, H_BLANCO = 21, H_RECUADRO = 6 * 21 + 2 * 24;
  let y = yInicio, pagina = 1;

  const finPagina = () => pagina * U - margen;
  const saltarPagina = () => {
    // Filas en blanco hasta pasar el corte de la página actual
    while (y <= pagina * U + 8) { filas.push({ tipo: 'blanco', h: H_BLANCO }); y += H_BLANCO; }
    pagina++;
  };

  d.items.forEach((it, i) => {
    const h = alturaItem_(it);
    if (y + h > finPagina()) {
      saltarPagina();
      filas.push({ tipo: 'enc', h: H_ENC }); y += H_ENC;
    }
    filas.push({ tipo: 'item', h: h, i: i }); y += h;
  });

  const separacion = CONFIG.FILAS_LIBRES * H_BLANCO;
  if (y + separacion + H_RECUADRO > finPagina()) {
    saltarPagina();                       // el recuadro empieza arriba de la última página
  } else {
    for (let b = 0; b < CONFIG.FILAS_LIBRES; b++) filas.push({ tipo: 'blanco', h: H_BLANCO });
  }
  return { filas: filas, paginas: pagina };
}

// Libera la zona de trabajo (columnas A:D, desde la fila 6) si está protegida.
// Si la protección pertenece a otra cuenta y este script no puede quitarla, explica qué hacer.
function liberarZonaProtegida_(hoja) {
  const L = CONFIG.FILA_LIMPIEZA;
  const yo = Session.getEffectiveUser().getEmail();
  const bloqueadas = [];

  // Protecciones de rangos que tocan la zona de trabajo
  hoja.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(pr => {
    if (pr.isWarningOnly()) return; // solo avisan, no bloquean
    const r = pr.getRange();
    if (r.getLastRow() < L || r.getColumn() > 4) return; // no toca la zona
    if (pr.canEdit()) pr.remove();
    else bloqueadas.push(r.getA1Notation());
  });

  // Protección de la hoja completa: se deja protegida, salvo la zona de trabajo
  const ph = hoja.getProtections(SpreadsheetApp.ProtectionType.SHEET)[0];
  if (ph && !ph.isWarningOnly()) {
    if (ph.canEdit()) {
      const zona = hoja.getRange(L, 1, hoja.getMaxRows() - L + 1, 4);
      ph.setUnprotectedRanges(ph.getUnprotectedRanges().concat([zona]));
    } else {
      bloqueadas.push('hoja completa');
    }
  }

  if (bloqueadas.length) {
    throw new Error('La hoja ' + CONFIG.HOJA_PRESUPUESTO + ' tiene protección que este script no puede quitar (' +
      bloqueadas.join(', ') + '). En Google Sheets abre Datos > Proteger hojas y rangos, quita esa protección ' +
      'o agrega la cuenta ' + yo + ' como editora.');
  }
}

// Ejecuta esta función desde el editor (botón ▶) para ver qué protecciones hay en la hoja.
function verProtecciones() {
  const hoja = hojaPresupuesto_();
  const yo = Session.getEffectiveUser().getEmail();
  Logger.log('El script corre como: ' + yo);
  [SpreadsheetApp.ProtectionType.SHEET, SpreadsheetApp.ProtectionType.RANGE].forEach(tipo => {
    hoja.getProtections(tipo).forEach(pr => {
      const editores = pr.getEditors().map(u => u.getEmail()).join(', ') || '(nadie)';
      Logger.log(tipo + ' | ' + (pr.getRange().getA1Notation()) + ' | solo aviso: ' + pr.isWarningOnly() +
        ' | yo puedo editarla: ' + pr.canEdit() + ' | editores: ' + editores + ' | ' + (pr.getDescription() || ''));
    });
  });
  Logger.log('Revisión terminada.');
}

// Detecta el formato real de la imagen leyendo sus primeros bytes (ignora lo que diga Drive)
function blobImagen_(bytes, nombre) {
  if (!bytes || bytes.length < 4) return null;
  const b = [bytes[0] & 0xff, bytes[1] & 0xff, bytes[2] & 0xff, bytes[3] & 0xff];
  if (b[0] === 0x89 && b[1] === 0x50 && b[2] === 0x4e && b[3] === 0x47) return Utilities.newBlob(bytes, 'image/png', nombre + '.png');
  if (b[0] === 0xff && b[1] === 0xd8) return Utilities.newBlob(bytes, 'image/jpeg', nombre + '.jpg');
  if (b[0] === 0x47 && b[1] === 0x49 && b[2] === 0x46) return Utilities.newBlob(bytes, 'image/gif', nombre + '.gif');
  return null; // WEBP, HEIC, PDF, etc. no los acepta la hoja
}

// Devuelve las versiones posibles del sello: primero una miniatura reducida (siempre PNG/JPG
// liviano) y de respaldo el archivo original.
function candidatosSello_() {
  const lista = [];
  try {
    const r = UrlFetchApp.fetch('https://drive.google.com/thumbnail?id=' + CONFIG.SELLO_ID + '&sz=w800', {
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      muteHttpExceptions: true
    });
    if (r.getResponseCode() === 200) {
      const bl = blobImagen_(r.getContent(), 'sello_miniatura');
      if (bl) lista.push(bl);
    }
  } catch (e) { Logger.log('Miniatura no disponible: ' + e.message); }
  try {
    const f = DriveApp.getFileById(CONFIG.SELLO_ID);
    const bl = blobImagen_(f.getBlob().getBytes(), 'sello_original');
    if (bl) lista.push(bl);
    else Logger.log('El archivo original no es PNG/JPG/GIF. Tipo en Drive: ' + f.getMimeType());
  } catch (e) { Logger.log('No puedo abrir el archivo del sello: ' + e.message); }
  return lista;
}

// Inserta el sello digital centrado (horizontal y vertical) sobre las filas del recuadro
// (49-53). El sello flota encima de la tabla; al ir al centro del ancho A:D queda entre
// «DATOS DEL CLIENTE» (izquierda) y «RESUMEN» (derecha) sin tapar el texto.
function insertarSello_(hoja, filaTop, nFilas) {
  const candidatos = candidatosSello_();
  if (!candidatos.length) throw new Error('No pude leer el sello como imagen PNG/JPG. Revisa el permiso y el formato del archivo en Drive.');

  let img = null, ultimoError = null;
  for (let i = 0; i < candidatos.length && !img; i++) {
    try { img = hoja.insertImage(candidatos[i], 1, filaTop); }
    catch (e) { ultimoError = e; }
  }
  if (!img) throw ultimoError;
  img.setAltTextTitle('SELLO_DIGITAL');

  // Alto disponible = suma de las alturas de las filas del recuadro
  let disp = 0;
  for (let r = 0; r < nFilas; r++) disp += hoja.getRowHeight(filaTop + r);

  // Escalar manteniendo la proporción (nunca más alto que el recuadro)
  const alto0 = img.getHeight(), ancho0 = img.getWidth();
  const alto = Math.min(alto0, CONFIG.SELLO_ALTO, disp - 8);
  const ancho = Math.round(ancho0 * alto / alto0);
  img.setHeight(alto).setWidth(ancho);

  // Centrar horizontalmente respecto al ancho A:D
  let total = 0;
  for (let c = 1; c <= 4; c++) total += hoja.getColumnWidth(c);
  let x = Math.max(0, Math.round((total - ancho) / 2));
  let col = 1;
  while (col < 4 && x >= hoja.getColumnWidth(col)) { x -= hoja.getColumnWidth(col); col++; }

  // Centrar verticalmente dentro de las filas del recuadro
  let y = Math.max(0, Math.round((disp - alto) / 2));
  let fila = filaTop;
  while (fila < filaTop + nFilas - 1 && y >= hoja.getRowHeight(fila)) { y -= hoja.getRowHeight(fila); fila++; }

  img.setAnchorCell(hoja.getRange(fila, col));
  img.setAnchorCellXOffset(x).setAnchorCellYOffset(y);
}

//  3. EXPORTAR A PDF (solo A:D y solo hasta la última fila usada)
// ---------------------------------------------------------------
function exportarPdf_(ultimaFila, numero) {
  const ss = ss_();
  const hoja = hojaPresupuesto_();
  DriveApp.getFileById(ss.getId()); // asegura el permiso de Drive para exportar

  // Formato carta: normalmente todo (filas 1 a 55) se ajusta a UNA sola página.
  // Si hay más de 36 artículos el documento crece y se ajusta al ancho (puede pasar a 2 páginas).
  const filasBase = CONFIG.FILA_RESUMEN + 7;   // 55: recuadro + dirección + teléfonos
  const ajuste = ultimaFila <= filasBase ? '&scale=4' : '&fitw=true'; // 4 = ajustar a la página

  const url = 'https://docs.google.com/spreadsheets/d/' + ss.getId() + '/export?format=pdf' +
    '&gid=' + hoja.getSheetId() +
    '&size=letter&portrait=true' + ajuste +
    '&gridlines=false&printtitle=false&sheetnames=false&pagenum=UNDEFINED&fzr=false' +
    '&top_margin=0.4&bottom_margin=0.4&left_margin=0.4&right_margin=0.4' +
    '&horizontal_alignment=CENTER' +
    '&r1=0&c1=0&r2=' + ultimaFila + '&c2=4';

  const resp = UrlFetchApp.fetch(url, {
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
    muteHttpExceptions: true
  });
  if (resp.getResponseCode() !== 200) throw new Error('Google no pudo exportar el PDF (código ' + resp.getResponseCode() + ').');
  const blob = resp.getBlob().setName('Presupuesto_' + numero + '.pdf');
  try {
    const paginas = (blob.getDataAsString('ISO-8859-1').match(/\/Type\s*\/Page[^s]/g) || []).length;
    Logger.log('Páginas reales del PDF: ' + paginas);
  } catch (e) {}
  return blob;
}

function crearPdfBlob_(d) {

function generarPdf(datos) {
  const d = normalizarDatos_(datos);
  const blob = crearPdfBlob_(d);
  return { nombre: blob.getName(), base64: Utilities.base64Encode(blob.getBytes()) };
}

function enviarPresupuestoPorCorreo(datos, correoDestino) {
  const correo = String(correoDestino || '').trim();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correo)) throw new Error('El correo "' + correo + '" no es válido.');
  const d = normalizarDatos_(datos);
  const opciones = opcionesRemitente_(); // valida el remitente antes de generar el PDF
  const blob = crearPdfBlob_(d);

  const cuerpo = 'Estimado/a ' + (d.cliente || 'cliente') + ',\n\n' +
    'Adjunto a este correo encontrará el presupuesto N° ' + d.numero + ' solicitado.\n\n' +
    'Quedamos a su entera disposición para cualquier duda o consulta.\n\n' +
    'Saludos cordiales,\n\n' +
    'Juan Carlos Romero F.P.\n' +
    'Técnico en Aire Acondicionado';

  opciones.attachments = [blob];
  GmailApp.sendEmail(correo, 'Presupuesto N° ' + d.numero + ' - ' + CONFIG.NOMBRE_REMITENTE, cuerpo, opciones);
  return 'EXITO';
}

// Decide desde qué dirección sale el correo
function opcionesRemitente_() {
  const remitente = String(CONFIG.CORREO_REMITENTE || '').trim().toLowerCase();
  const propio = String(Session.getEffectiveUser().getEmail() || '').toLowerCase();
  const opciones = { name: CONFIG.NOMBRE_REMITENTE, replyTo: remitente || propio };
  if (CONFIG.COPIA_OCULTA) opciones.bcc = CONFIG.COPIA_OCULTA;

  if (!remitente || remitente === propio) return opciones; // sale de la cuenta dueña del script

  const alias = GmailApp.getAliases().map(a => a.toLowerCase());
  if (alias.indexOf(remitente) === -1) {
    throw new Error('No puedo enviar desde ' + CONFIG.CORREO_REMITENTE + '. Este script pertenece a ' + propio +
      '. Cambia CORREO_REMITENTE a ' + propio + ', o agrega ' + CONFIG.CORREO_REMITENTE +
      ' en Gmail > Configuración > Cuentas > "Enviar como".');
  }
  opciones.from = remitente;
  return opciones;
}

}

// ---------------------------------------------------------------
//  RECUPERAR CLAVE (botón "¿Olvidaste tu clave?" de la app)
//  Envía la clave al mismo correo desde el que salen los presupuestos.
// ---------------------------------------------------------------
function recuperarClave_() {
  // Evita abusos: máximo un correo cada 5 minutos
  const cache = CacheService.getScriptCache();
  if (cache.get('recuperacion_enviada')) {
    throw new Error('Ya se envió un correo hace poco. Revisa tu bandeja (y spam) o intenta en 5 minutos.');
  }

  const propio = String(Session.getEffectiveUser().getEmail() || '').trim();
  const remitente = String(CONFIG.CORREO_REMITENTE || '').trim();
  // Si CORREO_REMITENTE sigue con el ejemplo o está vacío, usa la cuenta dueña del script
  const destino = (remitente && remitente.toLowerCase() !== 'Luxdaris7@gmail.com') ? remitente : propio;

  let opciones;
  try { opciones = opcionesRemitente_(); }
  catch (err) { opciones = { name: CONFIG.NOMBRE_REMITENTE }; } // si el alias falla, sale desde la cuenta dueña
  delete opciones.bcc; // la clave no debe ir en copia oculta

  const clave = escaparHtml_(CONFIG.CLAVE_ACCESO);
  opciones.htmlBody =
    '<div style="font-family:Arial,sans-serif;max-width:480px;margin:auto;padding:24px;border:1px solid #e2e8f0;border-radius:8px">' +
      '<h2 style="color:#0b111a;margin-top:0">🔑 Recuperación de clave</h2>' +
      '<p>Se pidió recuperar la clave de acceso de <b>Presupuestos · ' + escaparHtml_(CONFIG.NOMBRE_REMITENTE) + '</b>.</p>' +
      '<p>Tu clave actual es:</p>' +
      '<p style="font-size:22px;font-weight:bold;background:#f1f5f9;padding:12px;text-align:center;border-radius:6px;letter-spacing:1px">' + clave + '</p>' +
      '<p style="color:#64748b;font-size:13px">Si no fuiste tú, cambia CLAVE_ACCESO en el Apps Script y vuelve a implementar.</p>' +
    '</div>';

  GmailApp.sendEmail(destino, '🔑 Recuperación de clave · Presupuestos',
    'Tu clave de acceso a Presupuestos es: ' + CONFIG.CLAVE_ACCESO, opciones);

  cache.put('recuperacion_enviada', '1', 300);
  return 'Te enviamos la clave a ' + ocultarCorreo_(destino) + '. Revisa también la carpeta de spam.';
}

// Muestra el correo parcialmente: ju***@gmail.com
function ocultarCorreo_(correo) {
  const partes = String(correo).split('@');
  return partes[0].slice(0, 2) + '***@' + (partes[1] || '');
}

function escaparHtml_(t) {
  return String(t).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

const ENCABEZADOS_HISTORIAL = ['Nº Presupuesto', 'Fecha', 'Cliente', 'Total $', 'Total Bs', 'RIF / CI', 'Detalle (no editar)'];

function hojaHistorial_() {
  const ss = ss_();
  let h = ss.getSheetByName(CONFIG.HOJA_HISTORIAL);
  if (!h) {
    h = ss.insertSheet(CONFIG.HOJA_HISTORIAL);
    h.appendRow(ENCABEZADOS_HISTORIAL);
  }
  // Completa encabezados si la hoja venía de la versión anterior (5 columnas)
  if (h.getMaxColumns() < 7) h.insertColumnsAfter(h.getMaxColumns(), 7 - h.getMaxColumns());
  h.getRange(1, 1, 1, 7).setValues([ENCABEZADOS_HISTORIAL])
    .setFontWeight('bold').setBackground('#1a2332').setFontColor('white');
  return h;
}

function buscarFilaHistorial_(h, numero) {
  const buscado = numeroEntero_(numero);
  if (h.getLastRow() < 2) return -1;
  const col = h.getRange(2, 1, h.getLastRow() - 1, 1).getValues();
  for (let i = 0; i < col.length; i++) {
    if (numeroEntero_(col[i][0]) === buscado) return i + 2;
  }
  return -1;
}

function guardarEnHistorial(datos) {
  const d = normalizarDatos_(datos);
  const t = calcularTotales_(d);
  return conBloqueo_(() => {
    const h = hojaHistorial_();
    const fila = ["'" + d.numero, new Date(), d.cliente, t.total, t.totalBs, d.rif, JSON.stringify(d)];
    let r = buscarFilaHistorial_(h, d.numero);
    if (r === -1) {
      h.appendRow(fila);
      r = h.getLastRow();
    } else {
      h.getRange(r, 1, 1, 7).setValues([fila]); // si ya existía ese Nº, se actualiza
    }
    h.getRange(r, 4).setNumberFormat(CONFIG.FMT_USD);
    h.getRange(r, 5).setNumberFormat(CONFIG.FMT_BS);
    return { proximo: proximoNumero_() };
  });
}

function obtenerHistorial() {
  const h = ss_().getSheetByName(CONFIG.HOJA_HISTORIAL);
  if (!h || h.getLastRow() < 2) return [];
  const cols = Math.min(7, h.getMaxColumns());
  const vals = h.getRange(2, 1, h.getLastRow() - 1, cols).getDisplayValues();
  return vals
    .filter(r => r[0] !== '')
    .map(r => ({
      numero: formatearNumero_(r[0]),
      fecha: r[1],
      cliente: r[2],
      totalUsd: r[3],
      totalBs: r[4],
      rif: r[5] || '',
      tieneDetalle: !!(r[6] && r[6].charAt(0) === '{')
    }))
    .reverse(); // más recientes primero
}

function obtenerDetalle(numero) {
  const h = ss_().getSheetByName(CONFIG.HOJA_HISTORIAL);
  if (!h) throw new Error('Todavía no existe la hoja Historial.');
  const r = buscarFilaHistorial_(h, numero);
  if (r === -1) throw new Error('No encontré el presupuesto Nº ' + formatearNumero_(numero) + '.');
  const json = h.getMaxColumns() >= 7 ? h.getRange(r, 7).getValue() : '';
  if (!json) throw new Error('El presupuesto Nº ' + formatearNumero_(numero) + ' se guardó con la versión anterior y no tiene el detalle de artículos, por eso no se puede reimprimir.');
  return JSON.parse(json);
}

function reimprimirPresupuesto(numero) {
  return generarPdf(obtenerDetalle(numero));
}

  }

function ordenarClientes_(h) {
  const ultima = h.getLastRow();
  if (ultima > 2) h.getRange(2, 1, ultima - 1, 6).sort({ column: 2, ascending: true });
}

function guardarClienteBD(c) {
  if (!c || !String(c.nombre || '').trim()) throw new Error('Falta el nombre del cliente.');
  return conBloqueo_(() => {
    const h = hojaClientes_();
    h.appendRow([new Date(), c.nombre, c.rif || '', c.telefono || '', c.correo || '', c.direccion || '']);
    ordenarClientes_(h);
    return 'EXITO';
  });
}

function obtenerClientes() {
  const h = ss_().getSheetByName(CONFIG.HOJA_CLIENTES);
  if (!h) return [];
  return h.getDataRange().getDisplayValues();
}

function actualizarCliente(fila, valores) {
  fila = Number(fila);
  if (!fila || fila < 2) throw new Error('Fila de cliente no válida.');
  if (!Array.isArray(valores) || valores.length !== 5) throw new Error('Datos de cliente incompletos.');
  return conBloqueo_(() => {
    const h = hojaClientes_();
    h.getRange(fila, 2, 1, 5).setValues([valores]);
    ordenarClientes_(h);
    return 'EXITO';
  });
}
